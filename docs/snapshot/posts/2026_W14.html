async function fetchAndDecrypt(filename) {
  try {
    const res = await fetch("../../dashboard/" + filename);
    if (!res.ok) return []; 
    const encryptedBuf = await res.arrayBuffer(); // Đã sửa lỗi rsev_gauge ở đây
    const keyBuf = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(secret_pass));
    const keyBytes = new Uint8Array(keyBuf);
    const dataBytes = new Uint8Array(encryptedBuf);
    for (let i = 0; i < dataBytes.length; i++) dataBytes[i] = dataBytes[i] ^ keyBytes[i % keyBytes.length];
    return JSON.parse(new TextDecoder().decode(dataBytes));
  } catch(e) { return []; }
}

getSmartVal = (dataList, wk, algoName) => {
  const rows = dataList.filter(d => Number(d.Week) === wk);
  if (!rows.length) return null;
  const row = rows[0];
  if (row[algoName] !== undefined) return row[algoName];
  if (row[algoName.toLowerCase()] !== undefined) return row[algoName.toLowerCase()];
  const matchRow = rows.find(r => r.Type === algoName);
  return matchRow ? matchRow.Threshold : null;
};

function countThresholds(disease, ward, yr, wk) {
  // 1. TÍNH SỐ CA THỰC TẾ TỪ CƠ SỞ DỮ LIỆU (KHÔNG LẤY TỪ GOOGLE SHEETS)
  let raw_data = disease === "Sốt xuất huyết" ? data_sxh : data_tcm;
  let cases = d3.sum(raw_data.filter(d => 
      d.Year === yr && 
      d.Week === wk && 
      (ward === "Toàn Thành phố" || d.NoiOHienTai_SauKhiSapNhap_WardId === ward)
  ), d => d.SoCa) || 0;

  // 2. LẤY CÁC NGƯỠNG DỊCH
  let w_data = data_threshold.filter(d => d.Disease === disease && d.NoiOHienTai_SauKhiSapNhap_WardId === ward && d.Year === yr);
  let sea = getSmartVal(w_data, wk, 'Seasonal') ?? 99999;
  let far = getSmartVal(w_data, wk, 'Farrington') ?? 99999;
  let cus = getSmartVal(w_data, wk, 'Cusum') ?? 99999;
  let cdc = getSmartVal(w_data, wk, 'CDC') ?? 99999; 
  
  // 3. CHẤM ĐIỂM CẢNH BÁO
  let score = 0; // Mặc định: An toàn
  if (cases > sea) {
    let over_count = 0;
    if (cases > far) over_count++;
    if (cases > cus) over_count++;
    if (cases > cdc) over_count++;

    if (over_count === 3) score = 3;      // Báo động
    else if (over_count === 2) score = 2; // Cảnh báo
    else if (over_count === 1) score = 1; // Lưu ý
  }
  return score;
}

function getDetailedHotspots(disease) {
  if(data_threshold.length === 0) return [];
  let wards = [...new Set(data_threshold.filter(d => d.Disease === disease && d.NoiOHienTai_SauKhiSapNhap_WardId !== "Toàn Thành phố").map(d => d.NoiOHienTai_SauKhiSapNhap_WardId))];
  let results = [];

  for(let w of wards) {
     let s0 = countThresholds(disease, w, sys_max_yr, sys_max_wk);
     let s1 = countThresholds(disease, w, sys_max_yr, sys_max_wk - 1);
     let s2 = countThresholds(disease, w, sys_max_yr, sys_max_wk - 2);
     
     // Chỉ xét điểm nóng nếu 3 tuần liên tiếp đều từ CẢNH BÁO (score >= 2) trở lên
     if (s0 >= 2 && s1 >= 2 && s2 >= 2) {
         let w_data = data_threshold.filter(d => d.Disease === disease && d.NoiOHienTai_SauKhiSapNhap_WardId === w && d.Year === sys_max_yr);
         let row = w_data.find(d => d.Week === sys_max_wk);
         let cases = row ? (row.cases || 0) : 0;
         
         // Nếu 3 tuần liên tiếp đều BÁO ĐỘNG (score === 3)
         if (s0 === 3 && s1 === 3 && s2 === 3) {
             results.push({ name: w.replace("Phường ", "P.").replace("Xã ", "X."), type: 3, cases: cases, color: '#dc3545' }); // Đỏ danger
         } else {
             results.push({ name: w.replace("Phường ", "P.").replace("Xã ", "X."), type: 2, cases: cases, color: '#fd7e14' }); // Cam warning
         }
     }
  }
  results.sort((a, b) => {
     if (b.type !== a.type) return b.type - a.type;
     return b.cases - a.cases;
  });
  return results;
}

function getCityKPI(raw_data, disease) {
  if(raw_data.length === 0) return { current: 0, trend: "---", status: {t: "KHÔNG CÓ DỮ LIỆU", c: "#999"} };
  let current = d3.sum(raw_data.filter(d => d.Year === sys_max_yr && d.Week === sys_max_wk), d => d.SoCa) || 0;
  let prev = d3.sum(raw_data.filter(d => d.Year === sys_max_yr && d.Week === sys_max_wk - 1), d => d.SoCa) || 0;
  
  let pct = prev > 0 ? ((current - prev) / prev * 100).toFixed(1) : 0;
  let trend = current >= prev ? `▲ tăng ${pct}%` : `▼ giảm ${Math.abs(pct)}%`;
  
  let score = countThresholds(disease, "Toàn Thành phố", sys_max_yr, sys_max_wk);
  let status = score === 0 ? {t: "Bình thường", c: "#28a745"} :
               score === 1 ? {t: "Lưu ý", c: "#ffc107"} :
               score === 2 ? {t: "Cảnh báo", c: "#fd7e14"} : 
                             {t: "Báo động", c: "#dc3545"};
                             
  return { current, trend, status };
}

function generateSummaryTable(disease_name, raw_data, threshold_data) {
  if (raw_data.length === 0) return "";
  
  // Màu sắc và ID theo từng bệnh
  let theme_color = disease_name === "Sốt xuất huyết" ? "#d35400" : "#27ae60";
  let unique_id = disease_name === "Sốt xuất huyết" ? "sxh" : "tcm";

  // Dữ liệu toàn thành phố từ raw_data (tính số ca hiện tại)
  let city_data = raw_data.filter(d => d.NoiOHienTai_SauKhiSapNhap_WardId === "Toàn Thành phố" || d.NoiOHienTai_SauKhiSapNhap_WardId === undefined);
  if (city_data.length === 0) city_data = raw_data;

  // 1. Số ca cộng dồn hiện tại (từ raw_data)
  let cum_curr = d3.sum(city_data.filter(d => d.Year === sys_max_yr && d.Week <= sys_max_wk), d => d.SoCa) || 0;
  
  // 2. Số ca cộng dồn CÙNG KỲ NĂM TRƯỚC (Lấy từ threshold_data)
  let thr_city_prev = threshold_data.filter(d => 
     d.Disease === disease_name && 
     d.NoiOHienTai_SauKhiSapNhap_WardId === "Toàn Thành phố" && 
     d.Year === sys_max_yr - 1 && 
     d.Week <= sys_max_wk
  );
  let cum_prev = d3.sum(thr_city_prev, d => d.cases) || 0;
  let diff_cum = cum_prev > 0 ? ((cum_curr - cum_prev) / cum_prev) * 100 : (cum_curr > 0 ? 100 : 0);

  // 3. Số ca tuần hiện tại (raw_data)
  let wk_curr = d3.sum(city_data.filter(d => d.Year === sys_max_yr && d.Week === sys_max_wk), d => d.SoCa) || 0;
  
  // 4. Số ca tuần trước (raw_data)
  let prev_yr = sys_max_wk === 1 ? sys_max_yr - 1 : sys_max_yr;
  let prev_wk = sys_max_wk === 1 ? 52 : sys_max_wk - 1;
  let wk_prev = d3.sum(city_data.filter(d => d.Year === prev_yr && d.Week === prev_wk), d => d.SoCa) || 0;
  let diff_wk = wk_prev > 0 ? ((wk_curr - wk_prev) / wk_prev) * 100 : (wk_curr > 0 ? 100 : 0);

  // 5. Trung bình 4 tuần trước (raw_data)
  let wk_list = [];
  for(let i=1; i<=4; i++) {
     let w = sys_max_wk - i;
     let y = sys_max_yr;
     if (w < 1) { w = 52 + w; y = y - 1; }
     wk_list.push(y * 100 + w);
  }
  let wk_4w_data = city_data.filter(d => wk_list.includes(d.Year * 100 + d.Week));
  let avg_4w = (d3.sum(wk_4w_data, d => d.SoCa) / 4) || 0;
  let diff_4w = avg_4w > 0 ? ((wk_curr - avg_4w) / avg_4w) * 100 : (wk_curr > 0 ? 100 : 0);

  // 6. Dữ liệu ngưỡng cảnh báo dịch (Dùng getSmartVal)
  let w_data_thr = threshold_data.filter(d => 
     d.Disease === disease_name && 
     d.NoiOHienTai_SauKhiSapNhap_WardId === "Toàn Thành phố" && 
     d.Year === sys_max_yr
  );

  let sea = getSmartVal(w_data_thr, sys_max_wk, 'Seasonal');
  let far = getSmartVal(w_data_thr, sys_max_wk, 'Farrington');

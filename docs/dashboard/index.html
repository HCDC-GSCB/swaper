    async function decryptData(url, password) {
      const response = await fetch(url);
      if (!response.ok) throw new Error("Không tìm thấy file dữ liệu: " + url);
      const encryptedBuf = await response.arrayBuffer();
      
      const keyBuf = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(password));
      const keyBytes = new Uint8Array(keyBuf);
      const dataBytes = new Uint8Array(encryptedBuf);

      for (let i = 0; i < dataBytes.length; i++) {
        dataBytes[i] = dataBytes[i] ^ keyBytes[i % keyBytes.length];
      }
      
      const jsonStr = new TextDecoder().decode(dataBytes);

// Viết lại hàm kpiContent để tính toán và hiển thị trend tăng/giảm
function kpiContent(title, current, previous = null) {
  let trendElement = "";

  if (previous !== null) {
    let pillStyle = "display: inline-block; background: rgba(255,255,255,0.95); padding: 4px 12px; border-radius: 20px; font-size: 0.85rem; font-weight: 800; margin-top: 10px; box-shadow: 0 2px 6px rgba(0,0,0,0.15);";
    
    if (previous === 0) {
      trendElement = current > 0 
        ? html`<div style="${pillStyle} color: #d32f2f;">▲ Tăng 100%</div>`
        : html`<div style="${pillStyle} color: #555555;">▬ Không đổi</div>`;
    } else {
      let pct_diff = ((current - previous) / previous) * 100;
      
      if (pct_diff > 0) {
        trendElement = html`<div style="${pillStyle} color: #d32f2f;">▲ Tăng ${pct_diff.toFixed(1)}%</div>`;
      } else if (pct_diff < 0) {
        trendElement = html`<div style="${pillStyle} color: #2e7d32;">▼ Giảm ${Math.abs(pct_diff).toFixed(1)}%</div>`;
      } else {
        trendElement = html`<div style="${pillStyle} color: #555555;">▬ 0%</div>`;
      }
    }
  } else {
    trendElement = html`<div></div>`; 
  }

  return html`
    <div style="position: relative; z-index: 10;">
      <h2 style="margin: 0; font-size: 2.8rem; font-weight: 900; color: #ffffff !important; text-shadow: 1px 1px 3px rgba(0,0,0,0.2);">${current}</h2>
      <p style="margin: 0; font-size: 1.05rem; text-transform: uppercase; font-weight: 700; color: #ffffff !important; opacity: 0.95;">${title}</p>
      ${trendElement}
    </div>
  `;
}

function prepareHighchartsData(data, categoryCol, customOrder = null) {
  const weeks = Array.from({length: 53}, (_, i) => i + 1);
  let categories = [...new Set(data.map(d => d[categoryCol]))];

  if (customOrder) {
    categories = customOrder.filter(c => categories.includes(c));
  } else {
    categories.sort();
  }
  
  return categories.map(cat => {
    const catData = weeks.map(wk => {
      const matchingRows = data.filter(d => d.Week === wk && d[categoryCol] === cat);
      return d3.sum(matchingRows, d => d.SoCa);
    });
    return { name: cat, data: catData };
  });
}

    // --- HÀM GIẢI MÃ XOR ---
    async function decryptData(url, password) {
      // 1. Tải file .dat về
      const response = await fetch(url);
      if (!response.ok) throw new Error("Không tìm thấy file dữ liệu!");
      const encryptedBuf = await response.arrayBuffer();
      
      // 2. Tạo key từ password
      const keyBuf = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(password));
      const keyBytes = new Uint8Array(keyBuf);
      const dataBytes = new Uint8Array(encryptedBuf);
      
      // 3. Giải mã XOR
      for (let i = 0; i < dataBytes.length; i++) {
        dataBytes[i] = dataBytes[i] ^ keyBytes[i % keyBytes.length];
      }
      
      // 4. Parse JSON
      const jsonStr = new TextDecoder().decode(dataBytes);
      return JSON.parse(jsonStr);
    }

    // --- HÀM XỬ LÝ CHÍNH ---
    async function handleLogin() {
      const pass = input.value;
      msg.textContent = "⏳ Đang tải và giải mã dữ liệu...";
      msg.style.color = "blue";
      
      try {
        // Kiểm tra Hash trước cho nhanh
        const passBuf = new TextEncoder().encode(pass);
        const hashBuf = await crypto.subtle.digest('SHA-256', passBuf);
        const hashHex = Array.from(new Uint8Array(hashBuf)).map(b => b.toString(16).padStart(2, '0')).join('');
        
        if (hashHex !== correct_hash) {
           throw new Error("Mật khẩu không đúng!");

        const [sxh, tcm] = await Promise.all([
          decryptData("sxh.dat", pass), // Tự động tìm trong cùng thư mục

        ]);

function prepareHighchartsData(data, categoryCol, customOrder = null) {
  const weeks = Array.from({length: 53}, (_, i) => i + 1);
  
  // Lấy danh sách các nhóm (category)
  let categories = [...new Set(data.map(d => d[categoryCol]))];
  
  // Sắp xếp nhóm
  if (customOrder) {
    categories = customOrder.filter(c => categories.includes(c)); 
  } else {
    categories.sort();
  }
  
  // Duyệt qua từng nhóm để tạo series
  return categories.map(cat => {
    const catData = weeks.map(wk => {
      // Thay vì dùng .find() (chỉ lấy 1 dòng), ta phải lọc tất cả các dòng
      // thuộc Tuần đó và Nhóm đó, sau đó tính tổng cột SoCa.
      const matchingRows = data.filter(d => d.Week === wk && d[categoryCol] === cat);
      return d3.sum(matchingRows, d => d.SoCa);
    });
    
    return { name: cat, data: catData };
  });
}

function createHighchart(id, options) {
  // Chỉ tạo 1 div trống để Highcharts vẽ vào
  // Quarto sẽ tự động bao div này trong 1 cái card có title xanh
  const container = html`<div id="${id}" style="width:100%; height:100%;"></div>`;
  
  const defaultOptions = {
    chart: { 
      type: 'column', 
      style: { fontFamily: 'Arial, sans-serif' },
      backgroundColor: 'transparent', // Để nền trong suốt tiệp với nền Card
      height: 400 
    },
    title: { text: null }, // Tắt title nội bộ của Highcharts
    xAxis: { 
      categories: Array.from({length: 53}, (_, i) => i + 1),
      title: { text: 'Tuần', style: { fontWeight: 'bold' } },
      crosshair: true
    },
    yAxis: { 
      min: 0, 
      title: { text: 'Số ca', style: { fontWeight: 'bold' } }
    },
    credits: { enabled: false },
    exporting: { enabled: true }
  };
  
  const finalOptions = { ...defaultOptions, ...options };
  
  // Vẽ biểu đồ sau khi render
  requestAnimationFrame(() => {
    if (Highcharts && document.getElementById(id)) {
        Highcharts.chart(document.getElementById(id), finalOptions);
    }
  });
  
  return container;
}

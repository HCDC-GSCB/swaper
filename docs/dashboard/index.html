    async function decryptData(url, password) {
      const response = await fetch(url);
      if (!response.ok) throw new Error("Không tìm thấy file dữ liệu: " + url);
      const encryptedBuf = await response.arrayBuffer();
      
      const keyBuf = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(password));
      const keyBytes = new Uint8Array(keyBuf);
      const dataBytes = new Uint8Array(encryptedBuf);

      for (let i = 0; i < dataBytes.length; i++) {
        dataBytes[i] = dataBytes[i] ^ keyBytes[i % keyBytes.length];
      }
      
      const jsonStr = new TextDecoder().decode(dataBytes);
      return JSON.parse(jsonStr);
    }

function prepareHighchartsData(data, categoryCol, customOrder = null) {
  const weeks = Array.from({length: 53}, (_, i) => i + 1);
  let categories = [...new Set(data.map(d => d[categoryCol]))];

  if (customOrder) {
    categories = customOrder.filter(c => categories.includes(c));
  } else {
    categories.sort();
  }
  
  return categories.map(cat => {
    const catData = weeks.map(wk => {
      const matchingRows = data.filter(d => d.Week === wk && d[categoryCol] === cat);
      return d3.sum(matchingRows, d => d.SoCa);
    });
    return { name: cat, data: catData };
  });
}

ranking_data = {
  // Kiểm tra dữ liệu đầu vào
  if (!data_threshold || data_threshold.length === 0) return [];

  // A. Lọc dữ liệu theo Bệnh & Năm
  const base_data = data_threshold.filter(d => 
    d.Year === sys_max_yr && 
    d.Disease === selected_disease &&
    d.NoiOHienTai_SauKhiSapNhap_WardId !== "Toàn Thành phố"
  );

  // B. Xử lý từng phường
  let processed_wards = [...new Set(base_data.map(d => d.NoiOHienTai_SauKhiSapNhap_WardId))].map(w_name => {
    const w_data = base_data.filter(d => d.NoiOHienTai_SauKhiSapNhap_WardId === w_name);
    
    const curr_row = w_data.find(d => d.Week === sys_max_wk);
    const cases_curr = curr_row ? (curr_row.cases || 0) : 0;
    const pop = curr_row ? (curr_row.danso || 0) : 0;
    const kv = curr_row ? (curr_row.KV || "Khác") : "Khác";
    
    // Cảnh báo
    const is_far = curr_row && curr_row.Farrington && cases_curr > curr_row.Farrington;
    const is_cus = curr_row && curr_row.Cusum && cases_curr > curr_row.Cusum;
    
    // Điểm số
    let warning_score = 0;
    if (is_far && is_cus) warning_score = 3;
    else if (is_far) warning_score = 2;
    else if (is_cus) warning_score = 1;

    // Chỉ số phụ
    const rate = pop > 0 ? (cases_curr / pop * 100000).toFixed(1) : "-";
    const prev_row = w_data.find(d => d.Week === sys_max_wk - 1);
    const cases_prev = prev_row ? (prev_row.cases || 0) : 0;
    let diff_prev = cases_prev > 0 ? ((cases_curr - cases_prev)/cases_prev*100) : (cases_curr > 0 ? 100 : 0);
    
    const prev_4w = w_data.filter(d => d.Week >= sys_max_wk - 4 && d.Week < sys_max_wk);
    const avg_4w = d3.mean(prev_4w, d => d.cases || 0) || 0;
    let diff_4w = avg_4w > 0 ? ((cases_curr - avg_4w)/avg_4w*100) : (cases_curr > 0 ? 100 : 0);

    return {
      name: w_name,
      KV: kv,
      cases: cases_curr,
      rate: rate,
      diff_prev_str: (diff_prev > 0 && cases_prev > 0 ? "+" : "") + diff_prev.toFixed(1) + "%",
      diff_4w_str: (diff_4w > 0 && avg_4w > 0 ? "+" : "") + diff_4w.toFixed(1) + "%",
      far_flag: is_far,
      cus_flag: is_cus,
      score: warning_score
    };
  });

  // C. Tính điểm Khu vực
  const kv_priority = d3.rollup(processed_wards, v => d3.sum(v, d => d.score), d => d.KV);

  // D. Lọc theo Sidebar
  let display_data = [];
  if (selected_ward === "Toàn Thành phố") {
    display_data = processed_wards;
  } else {
    const target_kv = processed_wards.find(d => d.name === selected_ward)?.KV;
    display_data = processed_wards.filter(d => d.KV === target_kv);
  }

  // E. Sắp xếp
  display_data.sort((a, b) => {
    const scoreA = kv_priority.get(a.KV) || 0;
    const scoreB = kv_priority.get(b.KV) || 0;
    if (scoreB !== scoreA) return scoreB - scoreA;
    if (b.score !== a.score) return b.score - a.score;
    return b.cases - a.cases;
  });

  return display_data;
}

    requestAnimationFrame(() => {
        Highcharts.mapChart(container, options);
    });
    return container;
  }

  // 4. VÒNG LẶP RENDER

      d.Year === sys_max_yr && 
      d.Disease === selected_disease &&
      d.NoiOHienTai_SauKhiSapNhap_WardId !== "Toàn Thành phố"
    );

    const data_series = map_data_source.map(d => {
      const cases = d.cases || 0; 
      const pop = d.danso || 1;
      const rate = (cases / pop * 100000); 
      const value = metricType === "Số ca mắc" ? cases : Number(rate.toFixed(1));
      
      return {
        "hc-key": d.NoiOHienTai_SauKhiSapNhap_WardId, 
        value: value,
        name: d.NoiOHienTai_SauKhiSapNhap_WardId,
        danso: pop
      };
    });

    // B. Tạo Container
    const chartContainer = html`<div style="height: 600px; width: 100%;"></div>`;

        chart: { map: topology },
        title: { text: null },
        credits: { enabled: false },

          stops: [

            [0.1, '#4e79a7'],     // Xanh
            [0.5, '#f28e2b'],     // Cam
            [1, '#e15759']        // Đỏ
          ]
        },

        mapNavigation: {
          enabled: true,
          buttonOptions: { verticalAlign: 'bottom' }
        },

        series: [{
          data: data_series,
          name: metricType,
          joinBy: ['name', 'hc-key'], // Khớp cột 'name' trong Map với 'hc-key' trong Data

    // --- HÀM GIẢI MÃ XOR ---
    async function decryptData(url, password) {
      const response = await fetch(url);
      if (!response.ok) throw new Error("Không tìm thấy file dữ liệu: " + url);
      const encryptedBuf = await response.arrayBuffer();
      
      const keyBuf = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(password));
      const keyBytes = new Uint8Array(keyBuf);
      const dataBytes = new Uint8Array(encryptedBuf);

      for (let i = 0; i < dataBytes.length; i++) {
        dataBytes[i] = dataBytes[i] ^ keyBytes[i % keyBytes.length];
      }
      
      const jsonStr = new TextDecoder().decode(dataBytes);
      return JSON.parse(jsonStr);
    }

    // --- HÀM XỬ LÝ CHÍNH ---
    async function handleLogin() {
      const pass = input.value;
      msg.textContent = "⏳ Đang tải và giải mã dữ liệu...";
      msg.style.color = "blue";

      try {
        // Kiểm tra Hash
        const passBuf = new TextEncoder().encode(pass);
        const hashBuf = await crypto.subtle.digest('SHA-256', passBuf);
        const hashHex = Array.from(new Uint8Array(hashBuf)).map(b => b.toString(16).padStart(2, '0')).join('');
        
        if (hashHex !== correct_hash) {
           throw new Error("Mật khẩu không đúng!");

        // Bắt đầu tải và giải mã 3 file (SXH, TCM, THRESHOLD)

// Hàm chuẩn bị data Highcharts (Generic)
function prepareHighchartsData(data, categoryCol, customOrder = null) {
  const weeks = Array.from({length: 53}, (_, i) => i + 1);
  let categories = [...new Set(data.map(d => d[categoryCol]))];

  if (customOrder) {
    categories = customOrder.filter(c => categories.includes(c));
  } else {
    categories.sort();
  }
  
  return categories.map(cat => {
    const catData = weeks.map(wk => {
      const matchingRows = data.filter(d => d.Week === wk && d[categoryCol] === cat);
      return d3.sum(matchingRows, d => d.SoCa);
    });
    return { name: cat, data: catData };
  });
}

{
  // --- A. CẤU HÌNH MÀU SẮC ---
  const col_cdc = '#2ca02c'; // Xanh lá
  const col_far = '#d62728'; // Đỏ
  const col_cus = '#ff7f0e'; // Cam
  const col_sea = '#000000'; // Đen

  const ref_map = {
    "Sốt xuất huyết": "2016, 2017, 2020, 2023, 2024",
    "Tay chân miệng": "2017, 2019, 2020, 2022, 2024"
  };
  const history_str = ref_map[selected_disease] || "";

  // --- B. TÌM TUẦN MỚI NHẤT CÓ DỮ LIỆU NGƯỠNG ---
  const available_weeks = chart_thr.map(d => Number(d.Week));
  const max_thr_wk = available_weeks.length > 0 ? Math.max(...available_weeks) : 53;

  // --- C. HÀM ĐỌC DỮ LIỆU THÔNG MINH ---
  const getSmartVal = (dataList, wk, algoName) => {
    if (wk > max_thr_wk) return null;
    const rows = dataList.filter(d => Number(d.Week) === wk);
    if (!rows.length) return null;
    const row = rows[0];

    // Check ưu tiên 1: Cấu trúc Wide (Chuẩn)
    if (row[algoName] !== undefined) return row[algoName];
    if (row[algoName.toLowerCase()] !== undefined) return row[algoName.toLowerCase()];
    // Check ưu tiên 2: Cấu trúc Long (Cũ)
    const matchRow = rows.find(r => r.Type === algoName);
    if (matchRow) return matchRow.Threshold;
    return null;
  };

  // --- D. CHUẨN BỊ DATA VẼ ---
  const weeks = Array.from({length: 53}, (_, i) => i + 1);

  // 1. Data Số ca mắc (Cột)
  const chart_cases = current_data.filter(d => {
    const ward_col = d["NoiOHienTai_SauKhiSapNhap_WardId"] || d["WardId"]; 
    const ward_match = (selected_ward === "Toàn Thành phố") || (ward_col === selected_ward);
    return ward_match && String(d.Year) === String(selected_year);
  });

  const series_cases = weeks.map(wk => {
    if (wk > max_thr_wk) return null; // Cắt dữ liệu tương lai
    const cases_in_week = chart_cases.filter(d => Number(d.Week) === wk);
    return d3.sum(cases_in_week, d => d.SoCa);
  });

  // 2. Data Ngưỡng (Đường) - Vẫn tính đủ để so sánh
  const line_far = weeks.map(wk => getSmartVal(chart_thr, wk, 'Farrington'));
  const line_cus = weeks.map(wk => getSmartVal(chart_thr, wk, 'Cusum')); 
  const line_cdc = weeks.map(wk => getSmartVal(chart_thr, wk, 'CDC')); 
  const line_sea = weeks.map(wk => getSmartVal(chart_thr, wk, 'Seasonal'));

  // --- E. TÍNH TOÁN VỊ TRÍ CẢNH BÁO ---
  const getPosPoint = (wk_idx, threshold_val, positionType) => {
      const actual = series_cases[wk_idx];
      if (actual === null || threshold_val === null) return null;
      if (actual <= threshold_val) return null; // Không vượt -> Không vẽ

      let y_pos;
      if (positionType === 1) y_pos = 1;                    // CUSUM: Đáy
      else if (positionType === 2) y_pos = actual / 2;      // Farrington: Giữa
      else if (positionType === 3) y_pos = actual;          // CDC: Đỉnh

      return {
          y: y_pos,
          actualValue: actual,
          thresholdValue: threshold_val
      };
  }

  const warn_cus_data = weeks.map((wk, i) => getPosPoint(i, line_cus[i], 1)); 
  const warn_far_data = weeks.map((wk, i) => getPosPoint(i, line_far[i], 2)); 
  const warn_cdc_data = weeks.map((wk, i) => getPosPoint(i, line_cdc[i], 3)); 


  // --- F. VẼ HIGHCHARTS ---
  return createHighchart("chart-threshold-final", {
    chart: { type: 'column' },
    title: { text: null },
    subtitle: {

        }

        }

    plotOptions: {
